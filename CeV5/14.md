Since these are client-side web applications hosted directly inside the AVIS-DL storage vaults, the actual server-side API script needs to accept those requests using JavaScript (running on a server-side engine like Node.js) rather than having clients call an external executable.
If we move the Mercwar ecosystem over from a PHP backend to a native JavaScript Open API architecture, here is how the literal endpoint file is written to handle anonymous multi-extension file uploads.

File Name: upload.js (The AVIS-DL Backend Ingestion API)

javascript
const express = require('express');
const multer = require('multer');
const path = require('path');
const fs = require('fs');

const app = express();
const PORT = 80; // Standard HTTP port to align with web origin serving

// Enforce strict asset ingestion validation parameters from NEXUS.MANIFEST
const ALLOWED_EXTENSIONS = ['.html', '.js', '.css', '.json'];
const ALLOWED_MIME_TYPES = ['text/html', 'application/javascript', 'text/css', 'application/json'];
const VAULT_STORAGE_DIR = path.join(__dirname, 'vault');

// Ensure the AVIS-DL local storage directory path physically exists
if (!fs.existsSync(VAULT_STORAGE_DIR)) {
    fs.mkdirSync(VAULT_STORAGE_DIR, { recursive: true });
}

// Configure storage disk behavior for incoming multi-tenant applications
const storageConfig = multer.diskStorage({
    destination: (req, file, callback) => {
        callback(null, VAULT_STORAGE_DIR);
    },
    filename: (req, file, callback) => {
        // Maintain the original file extension so it serves cleanly back to browsers
        const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
        callback(null, file.fieldname + '-' + uniqueSuffix + path.extname(file.originalname));
    }
});

// Enforce schema compliance boundaries during stream parsing
const fileFilterRules = (req, file, callback) => {
    const fileExtension = path.extname(file.originalname).toLowerCase();
    const fileMimeType = file.mimetype.toLowerCase();

    const isExtensionValid = ALLOWED_EXTENSIONS.includes(fileExtension);
    const isMimeValid = ALLOWED_MIME_TYPES.includes(fileMimeType);

    if (isExtensionValid && isMimeValid) {
        callback(null, true); // Token verified, accept stream payload
    } else {
        // Trigger structured schema validation exceptions natively
        callback(new Error(`SchemaComplianceException: File extension '${fileExtension}' or MIME type '${fileMimeType}' is prohibited by NEXUS.MANIFEST rule layers.`), false);
    }
};

const uploadProcessor = multer({ 
    storage: storageConfig,
    fileFilter: fileFilterRules,
    limits: { fileSize: 5 * 1024 * 1024 } // Lock maximum file boundaries to 5MB
}).single('app_file'); // Listens precisely for the form field name specified in client frontends

/* ==============================================================================
   AVIS-DL OPEN API ROUTING DEFINITIONS
   ============================================================================== */

// Root Dashboard Acknowledgment Vector
app.get('/', (req, res) => {
    res.status(200).send("MERCWAR OPEN API ENGINE: AVIS-DL RUNTIME CORE IS ACTIVE AND SERVED.");
});

// Asset Ingestion Endpoint
app.post('/AVIS-DL/upload.php', (req, res) => {
    uploadProcessor(req, res, (error) => {
        if (error) {
            // Handle parsing violations or file boundary overflows
            return res.status(400).json({
                status: "failure",
                error_class: "SchemaComplianceException",
                diagnostic_message: error.message
            });
        }

        if (!req.file) {
            return res.status(400).json({
                status: "failure",
                diagnostic_message: "InvalidPayloadException: Missing multi-part data payload element 'app_file'."
            });
        }

        // Return standardized structural metadata for the compiled asset location
        return res.status(201).json({
            status: "success",
            resource_path: `/AVIS-DL/vault/${req.file.filename}`,
            serving_origin: `${req.protocol}://${req.get('host')}`,
            access: "public_served",
            telemetry: {
                bytes_ingested: req.file.size,
                original_identity: req.file.originalname
            }
        });
    });
});

// Start the network listening matrix
app.listen(PORT, () => {
    console.log(`================================================================================`);
    console.log(`MERCWAR DISTRIBUTED OPEN API SUB-SYSTEM INITIALIZED`);
    console.log(`================================================================================`);
    console.log(`Gateway Listening active on port: ${PORT}`);
    console.log(`Ingestion Vault Target Node path: ${VAULT_STORAGE_DIR}`);
});
